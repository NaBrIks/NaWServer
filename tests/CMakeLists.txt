function(add_protocol_test name source library)
    add_executable(${name}_test ${source})
    target_include_directories(${name}_test PRIVATE ../src)
    target_link_libraries(${name}_test PRIVATE ${library})
    set_target_properties(${name}_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${name}_test PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus)
    else()
        target_compile_options(${name}_test PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
    set_tests_properties(${name} PROPERTIES TIMEOUT 45)
endfunction()
