cmake_minimum_required(VERSION 3.16)
project(NaWServer_host LANGUAGES CXX)
include(../cmake/NaWServerConfiguration.cmake)
find_package(Threads REQUIRED)
if(WIN32)
    add_compile_definitions(NOMINMAX)
endif()
option(NAWSERVER_MCP "Enable the bounded embedded MCP endpoint in host demos" OFF)
option(NAWSERVER_HOST_TCP_MANAGED "Compile managed HTTP buffers and keep-alive" ON)
option(NAWSERVER_JSON_BOUNDED "Select the bounded exact-integer JSON backend" OFF)
option(NAWSERVER_JSON_YYJSON "Select the optional yyjson reader/validator" OFF)
option(NAWSERVER_JSON_RAPIDJSON "Select the optional RapidJSON SAX reader/validator" OFF)
set(NAWSERVER_YYJSON_TARGET "" CACHE STRING "Reuse an existing compatible yyjson CMake target")
set(NAWSERVER_RAPIDJSON_TARGET "" CACHE STRING "Reuse an existing compatible RapidJSON header target")
if(NAWSERVER_MCP)
    set(nawserver_json_choices 0)
    foreach(reader BOUNDED YYJSON RAPIDJSON)
        if(NAWSERVER_JSON_${reader})
            math(EXPR nawserver_json_choices "${nawserver_json_choices}+1")
        endif()
    endforeach()
    if(NOT nawserver_json_choices EQUAL 1)
        message(FATAL_ERROR "Select exactly one NaWServer JSON backend")
    elseif(NAWSERVER_JSON_BOUNDED)
        set(nawserver_json_backend bounded)
    elseif(NAWSERVER_JSON_YYJSON)
        set(nawserver_json_backend yyjson)
        enable_language(C)
    elseif(NAWSERVER_JSON_RAPIDJSON)
        set(nawserver_json_backend rapidjson)
    else()
        message(FATAL_ERROR "NAWSERVER_MCP requires exactly one JSON backend")
    endif()
endif()
get_filename_component(compiler_name "${CMAKE_CXX_COMPILER}" NAME)
if(compiler_name MATCHES "^zig(\\.exe)?$")
    set(CMAKE_CXX_ARCHIVE_CREATE "\"${CMAKE_CXX_COMPILER}\" ar qc <TARGET> <OBJECTS>")
    set(CMAKE_CXX_ARCHIVE_FINISH "\"${CMAKE_CXX_COMPILER}\" ranlib <TARGET>")
endif()
include(../cmake/NaWServerSources.cmake)
option(NAWSERVER_API_KEYS "Build persistent API keys" ON)
option(NAWSERVER_AUTH "Build authentication providers" ON)
option(NAWSERVER_CLIENT_QUOTAS "Build client quota policies" ON)
nawserver_collect_sources(nawserver_http_sources CORE WS SSE SSE_QUEUED)
if(NAWSERVER_AUTH)
    nawserver_collect_sources(nawserver_auth_sources AUTH)
    list(APPEND nawserver_http_sources ${nawserver_auth_sources})
endif()
if(NAWSERVER_API_KEYS)
    nawserver_collect_sources(nawserver_key_sources API_KEYS)
    list(APPEND nawserver_http_sources ${nawserver_key_sources} ../src/api_keys/file.cpp)
endif()
add_library(nawserver_http STATIC ${nawserver_http_sources})
target_include_directories(nawserver_http PUBLIC ../include PRIVATE ../src)
target_compile_features(nawserver_http PUBLIC cxx_std_17)
target_compile_definitions(nawserver_http PUBLIC NAWSERVER_API_KEYS=$<BOOL:${NAWSERVER_API_KEYS}>
    NAWSERVER_AUTH=$<BOOL:${NAWSERVER_AUTH}>
    NAWSERVER_CLIENT_QUOTAS=$<BOOL:${NAWSERVER_CLIENT_QUOTAS}>)
# Disabled builds compile no MCP/JSON source and allocate no MCP workspace.
if(NAWSERVER_MCP)
    include(../cmake/NaWServerJson.cmake)
    nawserver_attach_json(nawserver_http "${nawserver_json_backend}")
    target_sources(nawserver_http PRIVATE ../src/mcp/mcp.cpp)
    target_compile_definitions(nawserver_http PUBLIC NAWSERVER_MCP=1)
    foreach(setting JSON_INPUT_BYTES JSON_STRING_BYTES JSON_CONTAINER_ITEMS JSON_DEPTH
            JSON_VALUE_CAPACITY JSON_WORKSPACE_BYTES MCP_OUTPUT_BYTES)
        set(NAWSERVER_${setting} "" CACHE STRING "Override the default ${setting} for this build")
        if(NOT "${NAWSERVER_${setting}}" STREQUAL "")
            target_compile_definitions(nawserver_http PUBLIC "NAWSERVER_${setting}=${NAWSERVER_${setting}}")
        endif()
    endforeach()
endif()
target_link_libraries(nawserver_http PUBLIC Threads::Threads)
if(WIN32 AND NAWSERVER_API_KEYS)
    target_link_libraries(nawserver_http PUBLIC bcrypt advapi32)
endif()
add_library(nawserver_memory STATIC ../src/backends/memory.cpp ../src/backends/memory_sse.cpp)
target_link_libraries(nawserver_memory PUBLIC nawserver_http)
target_include_directories(nawserver_memory PRIVATE ../src)
foreach(target nawserver_http nawserver_memory)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endforeach()
