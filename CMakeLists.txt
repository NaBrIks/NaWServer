# Standalone consumers use the same source registry as the ESP-IDF component.
include("${CMAKE_CURRENT_LIST_DIR}/cmake/NaWServerConfiguration.cmake")
if(NOT COMMAND idf_component_register)
    cmake_minimum_required(VERSION 3.20)
    project(NaWServer LANGUAGES CXX)
    option(NAWSERVER_WEBSOCKET "Build WebSocket support" ON)
    option(NAWSERVER_SSE "Build SSE support" ON)
    option(NAWSERVER_TCP_MANAGED "Build managed HTTP buffers and keep-alive" ON)
    option(NAWSERVER_API_KEYS "Build persistent API key storage" ON)
    option(NAWSERVER_AUTH "Build authentication providers" ON)
    option(NAWSERVER_CLIENT_QUOTAS "Build client quota policies" ON)
    set(NAWSERVER_PLATFORM "" CACHE STRING "windows, linux or pico; host is auto-detected")
    include("${CMAKE_CURRENT_LIST_DIR}/cmake/NaWServer.cmake")
    set(nawserver_arguments "")
    if(NOT NAWSERVER_WEBSOCKET)
        list(APPEND nawserver_arguments NO_WEBSOCKET)
    endif()
    if(NOT NAWSERVER_SSE)
        list(APPEND nawserver_arguments NO_SSE)
    endif()
    if(NAWSERVER_TCP_MANAGED)
        list(APPEND nawserver_arguments MANAGED)
    endif()
    if(NOT NAWSERVER_AUTH)
        list(APPEND nawserver_arguments NO_AUTH)
    endif()
    if(NOT NAWSERVER_CLIENT_QUOTAS)
        list(APPEND nawserver_arguments NO_CLIENT_QUOTAS)
    endif()
    if(NOT NAWSERVER_API_KEYS)
        list(APPEND nawserver_arguments NO_API_KEYS)
    endif()
    nawserver_add_library(nawserver PLATFORM "${NAWSERVER_PLATFORM}" ${nawserver_arguments})
    add_library(NaWServer::NaWServer ALIAS nawserver)
    return()
endif()
# One HTTP engine on every platform. IDF owns the socket/RTOS implementation;
# no HTTPD-specific parser, lifecycle, WebSocket or TLS dependency is selected.
include("${CMAKE_CURRENT_LIST_DIR}/cmake/NaWServerSources.cmake")
nawserver_collect_sources(nawserver_sources CORE HTTP_WIRE TCP_SESSION TCP_REGISTRY TCP_RUNTIME)
list(APPEND nawserver_sources
    src/backends/tcp/platform/idf_tcp.cpp src/backends/tcp/platform/idf_worker.cpp)
if(CONFIG_NAWSERVER_AUTH)
    nawserver_collect_sources(nawserver_part AUTH)
    list(APPEND nawserver_sources ${nawserver_part})
endif()
if(CONFIG_NAWSERVER_TCP_MANAGED)
    nawserver_collect_sources(nawserver_part TCP_MANAGED)
    list(APPEND nawserver_sources ${nawserver_part})
endif()
if(CONFIG_NAWSERVER_WEBSOCKET)
    nawserver_collect_sources(nawserver_part WS WS_WIRE TCP_WS)
    list(APPEND nawserver_sources ${nawserver_part})
endif()
if(CONFIG_NAWSERVER_SSE)
    nawserver_collect_sources(nawserver_part SSE SSE_QUEUED TCP_SSE)
    list(APPEND nawserver_sources ${nawserver_part})
endif()
set(nawserver_private_requires freertos lwip esp_timer nvs_flash esp_hw_support)
if(CONFIG_NAWSERVER_API_KEYS)
    nawserver_collect_sources(nawserver_key_sources API_KEYS)
    list(APPEND nawserver_sources ${nawserver_key_sources} src/api_keys/idf.cpp)
endif()
idf_component_register(
    SRCS ${nawserver_sources}
    INCLUDE_DIRS include
    PRIV_INCLUDE_DIRS src
    PRIV_REQUIRES ${nawserver_private_requires}
)
target_compile_definitions(${COMPONENT_LIB} PUBLIC
    NAWSERVER_API_KEYS=$<BOOL:${CONFIG_NAWSERVER_API_KEYS}>
    NAWSERVER_AUTH=$<BOOL:${CONFIG_NAWSERVER_AUTH}>
    NAWSERVER_CLIENT_QUOTAS=$<BOOL:${CONFIG_NAWSERVER_CLIENT_QUOTAS}>
    NAWSERVER_TCP_MANAGED=$<BOOL:${CONFIG_NAWSERVER_TCP_MANAGED}>)

target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
if(CONFIG_NAWSERVER_TCP_MANAGED)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC
        NAWSERVER_TCP_MANAGED_HEAD_BYTES=${CONFIG_NAWSERVER_TCP_MANAGED_HEAD_BYTES}
        NAWSERVER_TCP_MANAGED_RESPONSE_BYTES=${CONFIG_NAWSERVER_TCP_MANAGED_RESPONSE_BYTES})
    if(CONFIG_NAWSERVER_WEBSOCKET)
        target_compile_definitions(${COMPONENT_LIB} PUBLIC
            NAWSERVER_TCP_MANAGED_STREAM_BYTES=${CONFIG_NAWSERVER_TCP_MANAGED_STREAM_BYTES})
    endif()
endif()
target_compile_options(${COMPONENT_LIB} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-std=gnu++17>)

if(CONFIG_NAWSERVER_MCP)
    set(nawserver_json_choices 0)
    foreach(reader BOUNDED YYJSON RAPIDJSON)
        if(CONFIG_NAWSERVER_JSON_${reader})
            math(EXPR nawserver_json_choices "${nawserver_json_choices}+1")
        endif()
    endforeach()
    if(NOT nawserver_json_choices EQUAL 1)
        message(FATAL_ERROR "NaWServer MCP requires exactly one JSON backend")
    elseif(CONFIG_NAWSERVER_JSON_BOUNDED)
        set(nawserver_json_backend bounded)
    elseif(CONFIG_NAWSERVER_JSON_YYJSON)
        set(nawserver_json_backend yyjson)
    elseif(CONFIG_NAWSERVER_JSON_RAPIDJSON)
        set(nawserver_json_backend rapidjson)
    else()
        message(FATAL_ERROR "NaWServer MCP requires a JSON backend")
    endif()
    include("${CMAKE_CURRENT_LIST_DIR}/cmake/NaWServerJson.cmake")
    nawserver_attach_json(${COMPONENT_LIB} "${nawserver_json_backend}")
    target_sources(${COMPONENT_LIB} PRIVATE src/mcp/mcp.cpp)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC NAWSERVER_MCP=1)
    foreach(setting JSON_INPUT_BYTES JSON_STRING_BYTES JSON_CONTAINER_ITEMS JSON_DEPTH
            JSON_VALUE_CAPACITY JSON_WORKSPACE_BYTES MCP_OUTPUT_BYTES)
        target_compile_definitions(${COMPONENT_LIB} PUBLIC "NAWSERVER_${setting}=${CONFIG_NAWSERVER_${setting}}")
    endforeach()
endif()
