cmake_minimum_required(VERSION 3.16)
project(NaWServer_mcp_checks LANGUAGES CXX)
enable_testing()
set(NAWSERVER_JSON_BACKEND "bounded" CACHE STRING "JSON reader: bounded, yyjson or rapidjson")
set_property(CACHE NAWSERVER_JSON_BACKEND PROPERTY STRINGS bounded yyjson rapidjson)
set(NAWSERVER_YYJSON_TARGET "" CACHE STRING "Existing compatible yyjson target")
if(NAWSERVER_JSON_BACKEND STREQUAL "yyjson")
    enable_language(C)
endif()
include(../../cmake/NaWServerJson.cmake)
foreach(target IN ITEMS mcp_checks demo_experiments_checks settings_checks)
    if(target STREQUAL "mcp_checks")
        set(corpus checks.cpp)
    elseif(target STREQUAL "demo_experiments_checks")
        set(corpus demo_experiments_checks.cpp)
    else()
        set(corpus settings_checks.cpp)
    endif()
    add_executable(${target} ${corpus} ../../src/mcp/mcp.cpp
        ../../src/http/http_server.cpp ../../src/http/web_request.cpp ../../src/http/web_response.cpp
        ../../src/http/form_parameters.cpp ../../src/http/url_decode.cpp)
    target_include_directories(${target} PRIVATE ../../include ../../src)
    target_compile_definitions(${target} PRIVATE NAWSERVER_MCP=1)
    target_compile_features(${target} PRIVATE cxx_std_17)
    nawserver_attach_json(${target} "${NAWSERVER_JSON_BACKEND}")
    if(target STREQUAL "settings_checks")
        target_compile_definitions(${target} PRIVATE NAWSERVER_JSON_INPUT_BYTES=8192
            NAWSERVER_JSON_STRING_BYTES=1024 NAWSERVER_JSON_DEPTH=12
            NAWSERVER_JSON_VALUE_CAPACITY=128 NAWSERVER_JSON_WORKSPACE_BYTES=32768)
    endif()
    foreach(setting JSON_INPUT_BYTES JSON_STRING_BYTES JSON_CONTAINER_ITEMS JSON_DEPTH
            JSON_VALUE_CAPACITY JSON_WORKSPACE_BYTES MCP_OUTPUT_BYTES)
        if(NOT target STREQUAL "settings_checks" AND DEFINED NAWSERVER_${setting}
                AND NOT "${NAWSERVER_${setting}}" STREQUAL "")
            target_compile_definitions(${target} PRIVATE "NAWSERVER_${setting}=${NAWSERVER_${setting}}")
        endif()
    endforeach()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
endforeach()
add_test(NAME mcp_json_contract COMMAND mcp_checks)
add_test(NAME mcp_demo_experiments COMMAND demo_experiments_checks)
add_test(NAME mcp_nondefault_settings COMMAND settings_checks)

# Merely selecting a reader cannot enable MCP or allocate its workspace.
add_executable(mcp_disabled_probe config_probe.cpp)
target_compile_definitions(mcp_disabled_probe PRIVATE NAWSERVER_JSON_BOUNDED=1)
add_test(NAME mcp_disabled_backend_does_not_enable COMMAND mcp_disabled_probe)
foreach(profile IN ITEMS missing_backend invalid_flag ambiguous_backend ambiguous_rapidjson unsupported_cjson
        invalid_depth invalid_input invalid_output invalid_values obsolete_prefix)
    if(profile STREQUAL obsolete_prefix)
        set(flags -DNaWServer_MCP=1 -DNAWSERVER_JSON_BOUNDED=1)
        set(expected "uppercase NAWSERVER_")
    elseif(profile STREQUAL missing_backend)
        set(flags -DNAWSERVER_MCP=1)
        set(expected "requires exactly one")
    elseif(profile STREQUAL invalid_flag)
        set(flags -DNAWSERVER_MCP=2 -DNAWSERVER_JSON_BOUNDED=1)
        set(expected "must have value 0 or 1")
    elseif(profile STREQUAL ambiguous_backend)
        set(flags -DNAWSERVER_MCP=1 -DNAWSERVER_JSON_BOUNDED=1 -DNAWSERVER_JSON_YYJSON=1)
        set(expected "requires exactly one")
    elseif(profile STREQUAL ambiguous_rapidjson)
        set(flags -DNAWSERVER_MCP=1 -DNAWSERVER_JSON_YYJSON=1 -DNAWSERVER_JSON_RAPIDJSON=1)
        set(expected "requires exactly one")
    elseif(profile STREQUAL unsupported_cjson)
        set(flags -DNAWSERVER_MCP=1 -DNAWSERVER_JSON_BOUNDED=1 -DNAWSERVER_MCP_CJSON=1)
        set(expected "cJSON does not implement")
    else()
        set(flags -DNAWSERVER_MCP=1 -DNAWSERVER_JSON_BOUNDED=1)
        if(profile STREQUAL invalid_depth)
            list(APPEND flags -DNAWSERVER_JSON_DEPTH=0)
        elseif(profile STREQUAL invalid_input)
            list(APPEND flags -DNAWSERVER_JSON_INPUT_BYTES=65536)
        elseif(profile STREQUAL invalid_output)
            list(APPEND flags -DNAWSERVER_MCP_OUTPUT_BYTES=255)
        else()
            list(APPEND flags -DNAWSERVER_JSON_VALUE_CAPACITY=0)
        endif()
        set(expected "NAWSERVER_.*(must|limits)")
    endif()
    try_compile(accepted "${CMAKE_BINARY_DIR}/config-${profile}"
        SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/config_probe.cpp"
        CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${CMAKE_CURRENT_SOURCE_DIR}/../../include"
        COMPILE_DEFINITIONS ${flags} OUTPUT_VARIABLE diagnostic)
    if(accepted OR NOT diagnostic MATCHES "${expected}")
        message(FATAL_ERROR "MCP invalid configuration ${profile} was not rejected clearly: ${diagnostic}")
    endif()
endforeach()
