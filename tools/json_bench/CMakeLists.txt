cmake_minimum_required(VERSION 3.16)
project(NaWServer_json_bench LANGUAGES C CXX)
set(NAWSERVER_JSON_BACKEND bounded CACHE STRING "bounded, yyjson or rapidjson")
option(NAWSERVER_BENCH_EXISTING_YYJSON "Model an application that already calls yyjson" OFF)
option(NAWSERVER_BENCH_EXISTING_RAPIDJSON "Model an application that already calls RapidJSON" OFF)
if(NAWSERVER_BENCH_EXISTING_YYJSON AND NAWSERVER_BENCH_EXISTING_RAPIDJSON)
    message(FATAL_ERROR "Select at most one existing application JSON consumer")
endif()
include(../../cmake/NaWServerJson.cmake)
add_executable(json_bench bench.cpp)
target_include_directories(json_bench PRIVATE ../../include)
target_compile_features(json_bench PRIVATE cxx_std_17)
target_compile_definitions(json_bench PRIVATE NAWSERVER_JSON_BENCH_HOST=1)
if(NAWSERVER_BENCH_EXISTING_YYJSON)
    # The same upstream object and options serve both the actual application
    # consumer and the selected adapter; this is not a second private copy.
    add_library(application_yyjson STATIC ../../third_party/yyjson/src/yyjson.c)
    target_include_directories(application_yyjson PUBLIC ../../third_party/yyjson/src)
    target_compile_definitions(application_yyjson PRIVATE YYJSON_DISABLE_FILE=1 YYJSON_DISABLE_UTILS=1
        YYJSON_DISABLE_NON_STANDARD=1 YYJSON_DISABLE_INCR_READER=1)
    set(NAWSERVER_YYJSON_TARGET application_yyjson)
    target_compile_definitions(json_bench PRIVATE NAWSERVER_BENCH_EXISTING_YYJSON=1)
    target_link_libraries(json_bench PRIVATE application_yyjson)
elseif(NAWSERVER_BENCH_EXISTING_RAPIDJSON)
    target_include_directories(json_bench SYSTEM PRIVATE ../../third_party/rapidjson/include)
    target_compile_definitions(json_bench PRIVATE NAWSERVER_BENCH_EXISTING_RAPIDJSON=1)
endif()
nawserver_attach_json(json_bench "${NAWSERVER_JSON_BACKEND}")
if(MSVC)
    target_compile_options(json_bench PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus /utf-8)
    target_link_options(json_bench PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/json_bench.map")
else()
    target_compile_options(json_bench PRIVATE -Wall -Wextra -Werror -pedantic)
    target_link_options(json_bench PRIVATE "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/json_bench.map")
endif()
