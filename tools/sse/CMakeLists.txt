cmake_minimum_required(VERSION 3.16)
project(NaWServer_sse_lab LANGUAGES CXX)
enable_testing()
add_executable(sse_test ../../src/buffer_memory.cpp ../../src/sse/encoder.cpp checks.cpp test_main.cpp)
target_include_directories(sse_test PRIVATE ../../include)
set_target_properties(sse_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(sse_test PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus /utf-8)
else()
    target_compile_options(sse_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME sse COMMAND sse_test)
add_executable(sse_bridge_test bridge_test.cpp ../../src/buffer_memory.cpp ../../src/sse/response_bridge.cpp ../../src/sse/encoder.cpp ../../src/http/web_response.cpp ../../src/routing/pipeline.cpp)
target_include_directories(sse_bridge_test PRIVATE ../../include ../../src)
set_target_properties(sse_bridge_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(sse_bridge_test PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus /utf-8)
else()
    target_compile_options(sse_bridge_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME sse_bridge COMMAND sse_bridge_test)
