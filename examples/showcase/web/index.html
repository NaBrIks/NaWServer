// Публичный WebSocket + защищённый HTTP
static WebSocketEndpoint ws("/ws");
static FixedBasicAuthenticator fixture("demo", "showcase-only");
static AuthMiddleware authCheck(fixture);
static AccessMiddleware accessCheck;
static auto chain = makeChain(authCheck, accessCheck);
static Route auth("/api/control", HTTP_POST, control_endpoint,
                  chain, AccessPolicy::authenticated());

ws.onMessage([](auto *client, const uint8_t *data,
               size_t len, bool binary) {
  if (binary) client->send_binary(data, len);
  else client->send(std::string((const char*)data, len).c_str());
});
server.addHandler(&ws);
server.addHandler(&auth);
server.begin();
