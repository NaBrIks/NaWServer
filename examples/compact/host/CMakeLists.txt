cmake_minimum_required(VERSION 3.20)
project(NaWServer_compact_host LANGUAGES CXX)
get_filename_component(NAWSERVER_ROOT "../../.." ABSOLUTE)
include("${NAWSERVER_ROOT}/cmake/NaWServer.cmake")
include("${NAWSERVER_ROOT}/cmake/NaWServerJson.cmake")
find_program(NAWSERVER_NODE node REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${NAWSERVER_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/../web/pack.mjs"
    "${CMAKE_CURRENT_BINARY_DIR}/generated/compact_web.hpp" COMMAND_ERROR_IS_FATAL ANY)
foreach(mode SSE WS)
    if(mode STREQUAL SSE)
        set(protocol NO_WEBSOCKET)
        set(websocket 0)
    else()
        set(protocol NO_SSE)
        set(websocket 1)
    endif()
    nawserver_add_library(compact_${mode} ${protocol} NO_AUTH NO_CLIENT_QUOTAS MANAGED)
    nawserver_attach_json(compact_${mode} bounded)
    target_compile_definitions(compact_${mode} PUBLIC NAWSERVER_API_KEYS_MAX=1 NAWSERVER_DEMO_STATUS_WS=${websocket})
    if(NOT MSVC)
        target_compile_options(compact_${mode} PRIVATE -ffunction-sections -fdata-sections)
    endif()
    add_executable(compact_${mode}_demo main.cpp ../common/demo.cpp)
    target_include_directories(compact_${mode}_demo PRIVATE ../common "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(compact_${mode}_demo PRIVATE compact_${mode})
    if(MSVC)
        target_compile_options(compact_${mode}_demo PRIVATE /W4 /WX)
    else()
        target_compile_options(compact_${mode}_demo PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
        target_link_options(compact_${mode}_demo PRIVATE -Wl,--gc-sections)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(compact_start_failure_test ../tests/start_failure_test.cpp ../common/demo.cpp)
    target_include_directories(compact_start_failure_test PRIVATE ../common "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(compact_start_failure_test PRIVATE compact_SSE)
    target_compile_definitions(compact_start_failure_test PRIVATE
        NAWSERVER_DEMO_TX_METADATA_MODE=1 NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES=2048)
    if(MSVC)
        target_compile_options(compact_start_failure_test PRIVATE /W4 /WX)
    else()
        target_compile_options(compact_start_failure_test PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
    endif()
    add_test(NAME compact_start_failure COMMAND compact_start_failure_test)
    set_tests_properties(compact_start_failure PROPERTIES TIMEOUT 15)
    add_test(NAME compact_host_accept
        COMMAND "${NAWSERVER_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/host-accept.mjs"
            "$<TARGET_FILE:compact_SSE_demo>")
    set_tests_properties(compact_host_accept PROPERTIES TIMEOUT 30)
endif()
