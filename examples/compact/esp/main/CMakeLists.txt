# Asset generation must not run during IDF's dependency-only script expansion:
# CMAKE_BINARY_DIR can be the source working directory in that phase.
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    find_program(NAWSERVER_DEMO_NODE node REQUIRED)
    set(NAWSERVER_DEMO_WEB_DIR "${CMAKE_CURRENT_LIST_DIR}/../../web")
    set(NAWSERVER_DEMO_WEB_HEADER "${CMAKE_BINARY_DIR}/generated/compact_web.hpp")
    execute_process(COMMAND "${NAWSERVER_DEMO_NODE}" "${NAWSERVER_DEMO_WEB_DIR}/pack.mjs" "${NAWSERVER_DEMO_WEB_HEADER}" RESULT_VARIABLE NAWSERVER_DEMO_PACK_RESULT)
    if(NOT NAWSERVER_DEMO_PACK_RESULT EQUAL 0)
        message(FATAL_ERROR "Compact web asset packing failed")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NAWSERVER_DEMO_WEB_DIR}/index.html" "${NAWSERVER_DEMO_WEB_DIR}/sha256.mjs" "${NAWSERVER_DEMO_WEB_DIR}/pack.mjs")
endif()
idf_component_register(SRCS main.cpp ../../platform/esp.cpp ../../common/demo.cpp
    INCLUDE_DIRS . ../../common ../../platform
    PRIV_REQUIRES NaWServer app_update esp_wifi esp_event esp_netif nvs_flash esp_timer esp_system esp_hw_support mbedtls esp_driver_usb_serial_jtag)

if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    set(NAWSERVER_DEMO_README_PROFILE "0" CACHE STRING "Optional measured README fixture: off/min/normal/max")
    if(NOT NAWSERVER_DEMO_README_PROFILE MATCHES "^[0-3]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_README_PROFILE must be 0..3")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NAWSERVER_DEMO_README_PROFILE=${NAWSERVER_DEMO_README_PROFILE})
    option(NAWSERVER_DEMO_ASYNC_PROBE "Optional periodic application/HTTP interference fixture" OFF)
    if(NAWSERVER_DEMO_ASYNC_PROBE AND NOT NAWSERVER_DEMO_README_PROFILE EQUAL 3)
        message(FATAL_ERROR "NAWSERVER_DEMO_ASYNC_PROBE requires NAWSERVER_DEMO_README_PROFILE=3")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NAWSERVER_DEMO_ASYNC_PROBE=$<BOOL:${NAWSERVER_DEMO_ASYNC_PROBE}>)
    set(NAWSERVER_DEMO_MEMORY_LINK_PROFILE "0" CACHE STRING "Link-only comparison: normal, network, HTTPD, plain HTTP, managed TX")
    if(NOT NAWSERVER_DEMO_MEMORY_LINK_PROFILE MATCHES "^[0-4]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_MEMORY_LINK_PROFILE must be 0..4")
    endif()
    if(NAWSERVER_DEMO_MEMORY_PROBE OR NAWSERVER_DEMO_MEMORY_BREAKDOWN OR NAWSERVER_DEMO_MEMORY_LINK_PROFILE EQUAL 2)
        idf_component_get_property(NAWSERVER_DEMO_HTTPD_LIB esp_http_server COMPONENT_LIB)
        target_link_libraries(${COMPONENT_LIB} PRIVATE ${NAWSERVER_DEMO_HTTPD_LIB})
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NAWSERVER_DEMO_MEMORY_BREAKDOWN=$<BOOL:${NAWSERVER_DEMO_MEMORY_BREAKDOWN}>)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NAWSERVER_DEMO_MEMORY_LINK_PROFILE=${NAWSERVER_DEMO_MEMORY_LINK_PROFILE})
    set(NAWSERVER_DEMO_TX_MEMORY_MODE "0" CACHE STRING "TX memory mode: lazy, preallocated heap or static")
    set(NAWSERVER_DEMO_TX_EXECUTOR "0" CACHE STRING "TX service: dedicated worker or shared existing workers")
    if(NOT NAWSERVER_DEMO_TX_EXECUTOR MATCHES "^[01]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_EXECUTOR must be 0 or 1")
    endif()
    set(NAWSERVER_DEMO_TX_METADATA_MODE "0" CACHE STRING "TX metadata: allocator, static arena or strict PSRAM")
    set(NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES "2048" CACHE STRING "Bytes reserved by the static TX metadata arena")
    set(NAWSERVER_DEMO_TX_PAYLOAD_PSRAM "0" CACHE STRING "Allocate TX payload strictly in PSRAM")
    set(NAWSERVER_DEMO_TX_INTERNAL_ONLY "1" CACHE STRING "Keep TCP/TX pools internal except explicit PSRAM overrides")
    set(NAWSERVER_DEMO_TX_BLOCKS "4" CACHE STRING "Number of TX blocks")
    set(NAWSERVER_DEMO_TX_BLOCK_BYTES "1024" CACHE STRING "Bytes per TX block")
    set(NAWSERVER_DEMO_HTTP_JOBS "2" CACHE STRING "Maximum concurrent HTTP jobs")
    set(NAWSERVER_DEMO_CONNECTIONS "4" CACHE STRING "Maximum managed connections")
    set(NAWSERVER_DEMO_INGRESS_SLOTS "2" CACHE STRING "Concurrent HTTP header ingress workspaces")
    set(NAWSERVER_DEMO_STACK_DIAGNOSTICS "0" CACHE STRING "Emit worker stack watermarks over USB")
    set(NAWSERVER_DEMO_MEMORY_SAMPLER "0" CACHE STRING "Emit internal/PSRAM memory samples once per second over USB")
    set(NAWSERVER_DEMO_FULL_STOP_PROBE "0" CACHE STRING "Probe complete Demo stop/restart after idle HTTP exchanges")
    set(NAWSERVER_DEMO_TX_BULK_PROBE "0" CACHE STRING "Enable bounded generated data through real TX payload leases")
    set(NAWSERVER_DEMO_NATIVE_ACCEPT_DIAGNOSTICS "0" CACHE STRING "Count native accept failures through a forwarding transport, sampled over UART")
    if(NOT NAWSERVER_DEMO_INGRESS_SLOTS MATCHES "^[0-9]+$" OR NAWSERVER_DEMO_INGRESS_SLOTS LESS 1 OR NAWSERVER_DEMO_INGRESS_SLOTS GREATER 12)
        message(FATAL_ERROR "NAWSERVER_DEMO_INGRESS_SLOTS must be 1..12")
    endif()
    if(NAWSERVER_DEMO_NATIVE_ACCEPT_DIAGNOSTICS AND NOT NAWSERVER_DEMO_MEMORY_SAMPLER)
        message(FATAL_ERROR "Native accept diagnostics require the memory sampler")
    endif()
    if(NAWSERVER_DEMO_TX_BULK_PROBE AND (NOT IDF_TARGET STREQUAL "esp32s3" OR NAWSERVER_DEMO_STATUS_WS))
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_BULK_PROBE requires the ESP32-S3 SSE profile")
    endif()
    if(NAWSERVER_DEMO_FULL_STOP_PROBE AND NOT (NAWSERVER_DEMO_MEMORY_PROBE OR NAWSERVER_DEMO_MEMORY_SAMPLER))
        message(FATAL_ERROR "NAWSERVER_DEMO_FULL_STOP_PROBE requires NAWSERVER_DEMO_MEMORY_PROBE or NAWSERVER_DEMO_MEMORY_SAMPLER")
    endif()
    if(NOT NAWSERVER_DEMO_TX_MEMORY_MODE MATCHES "^[012]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_MEMORY_MODE must be 0, 1 or 2")
    endif()
    if(NOT NAWSERVER_DEMO_TX_METADATA_MODE MATCHES "^[012]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_METADATA_MODE must be 0, 1 or 2")
    endif()
    if(NOT NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES MATCHES "^[0-9]+$" OR NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES LESS 256 OR NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES GREATER 65536)
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES must be 256..65536")
    endif()
    if(NOT NAWSERVER_DEMO_TX_PAYLOAD_PSRAM MATCHES "^[01]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_PAYLOAD_PSRAM must be 0 or 1")
    endif()
    if(NOT NAWSERVER_DEMO_TX_INTERNAL_ONLY MATCHES "^[01]$")
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_INTERNAL_ONLY must be 0 or 1")
    endif()
    if(NAWSERVER_DEMO_TX_METADATA_MODE EQUAL 2 OR NAWSERVER_DEMO_TX_PAYLOAD_PSRAM)
        if(NOT IDF_TARGET STREQUAL "esp32s3" OR NOT CONFIG_SPIRAM)
            message(FATAL_ERROR "PSRAM metadata/payload requires ESP32-S3 with CONFIG_SPIRAM=y")
        endif()
    endif()
    if(NAWSERVER_DEMO_TX_MEMORY_MODE EQUAL 2 AND NAWSERVER_DEMO_TX_PAYLOAD_PSRAM)
        message(FATAL_ERROR "Static TX payload cannot be combined with PSRAM TX payload")
    endif()
    if(NAWSERVER_DEMO_TX_BLOCK_BYTES LESS 768 OR NAWSERVER_DEMO_TX_BLOCK_BYTES GREATER 4096)
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_BLOCK_BYTES must be 768..4096")
    endif()
    math(EXPR NAWSERVER_DEMO_TX_ALIGNMENT_REMAINDER "${NAWSERVER_DEMO_TX_BLOCK_BYTES} % 16")
    if(NOT NAWSERVER_DEMO_TX_ALIGNMENT_REMAINDER EQUAL 0)
        message(FATAL_ERROR "NAWSERVER_DEMO_TX_BLOCK_BYTES must be a multiple of 16")
    endif()
    target_compile_features(${COMPONENT_LIB} PRIVATE cxx_std_17)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE NAWSERVER_DEMO_TX_MEMORY_MODE=${NAWSERVER_DEMO_TX_MEMORY_MODE}
        NAWSERVER_DEMO_TX_EXECUTOR=${NAWSERVER_DEMO_TX_EXECUTOR}
        NAWSERVER_DEMO_TX_METADATA_MODE=${NAWSERVER_DEMO_TX_METADATA_MODE} NAWSERVER_DEMO_TX_PAYLOAD_PSRAM=${NAWSERVER_DEMO_TX_PAYLOAD_PSRAM}
        NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES=${NAWSERVER_DEMO_TX_METADATA_STATIC_BYTES}
        NAWSERVER_DEMO_TX_INTERNAL_ONLY=${NAWSERVER_DEMO_TX_INTERNAL_ONLY}
        NAWSERVER_DEMO_TX_BLOCKS=${NAWSERVER_DEMO_TX_BLOCKS} NAWSERVER_DEMO_TX_BLOCK_BYTES=${NAWSERVER_DEMO_TX_BLOCK_BYTES}
        NAWSERVER_DEMO_HTTP_JOBS=${NAWSERVER_DEMO_HTTP_JOBS} NAWSERVER_DEMO_CONNECTIONS=${NAWSERVER_DEMO_CONNECTIONS}
        NAWSERVER_DEMO_INGRESS_SLOTS=${NAWSERVER_DEMO_INGRESS_SLOTS}
        NAWSERVER_DEMO_NATIVE_ACCEPT_DIAGNOSTICS=$<BOOL:${NAWSERVER_DEMO_NATIVE_ACCEPT_DIAGNOSTICS}>
        NAWSERVER_DEMO_STACK_DIAGNOSTICS=${NAWSERVER_DEMO_STACK_DIAGNOSTICS} NAWSERVER_DEMO_MEMORY_PROBE=$<BOOL:${NAWSERVER_DEMO_MEMORY_PROBE}>
        NAWSERVER_DEMO_MEMORY_SAMPLER=$<BOOL:${NAWSERVER_DEMO_MEMORY_SAMPLER}>
        NAWSERVER_DEMO_FULL_STOP_PROBE=$<BOOL:${NAWSERVER_DEMO_FULL_STOP_PROBE}>
        NAWSERVER_DEMO_TX_BULK_PROBE=$<BOOL:${NAWSERVER_DEMO_TX_BULK_PROBE}>)
    target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_BINARY_DIR}/generated")
    target_compile_options(${COMPONENT_LIB} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti;-fstack-usage>)
    include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/NaWServerJson.cmake")
    nawserver_attach_json(${COMPONENT_LIB} bounded)
    if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../.private")
        target_include_directories(${COMPONENT_LIB} PRIVATE ../.private)
    endif()
endif()
