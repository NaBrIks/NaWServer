cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w)
set(PICO_PLATFORM rp2350-arm-s)
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(CompactRadioScan LANGUAGES C CXX ASM)
pico_sdk_init()
add_executable(compact_radio_scan main.cpp)
target_include_directories(compact_radio_scan PRIVATE ../../.private)
target_link_libraries(compact_radio_scan pico_stdlib pico_cyw43_arch_none)
pico_set_binary_type(compact_radio_scan no_flash)
pico_enable_stdio_usb(compact_radio_scan 1)
pico_enable_stdio_uart(compact_radio_scan 0)
pico_add_extra_outputs(compact_radio_scan)
